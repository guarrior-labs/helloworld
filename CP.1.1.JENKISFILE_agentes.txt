pipeline {
    agent none
    stages {

        stage('Get Code') {
            agent any

            steps {
                echo '=== LIMPIANDO WORKSPACE ==='
                deleteDir()

                echo '=== DESCARGANDO REPOSITORIO ==='
                git 'https://github.com/guarrior-labs/helloworld.git'

                echo '=== COMPARTIR ARCHIVOS CON DEMAS AGENTES ==='
                stash name: 'source-code', includes: '**/*'
            }
        }

        stage('Tests') {
            parallel {

                stage('UnitCover') {
                    agent { label 'Nodo1' }
                    steps {

                        echo '==- RECUPERANDO ARCHIVOS NODO 1 ==-'
                        unstash 'source-code'
                        catchError(buildResult: 'UNSTABLE', stageResult: 'FAILURE') {
                            bat '''
                                echo ==- CONF. PYTHONPATH NODO 1==-
                                set "PYTHONPATH=%WORKSPACE%"

                                echo ==- DATOS NODO 1 ==-
                                hostname
                                whoami
                                echo WORKSPACE=%WORKSPACE%

                                echo ==- VERSIONES MODULOS NODO 1 ==-
                                java --version
                                python --version
                                python -m pytest --version

                                echo ==- INSTALA MODULOS NODO 1 -==
                                python -m pip install --disable-pip-version-check -U coverage pytest

                                echo ===== EJECUTA UNITARIAS/COVERAGE NODO 1 =====
                                coverage run --branch --source=app --omit=app\\__init__.py,app\\api.py -m pytest --junitxml=result-unit.xml test\\unit
                                coverage xml
                            '''
                        }

                        echo '==- PUBLICA RESULTADOS NODO 1 ==-'
                        junit 'result-unit.xml'

                        recordCoverage(
                            tools: [[
                                parser: 'COBERTURA',
                                pattern: '**/coverage.xml'
                            ]],                        
                            qualityGates: [
                                // Cobertura de líneas:
                                // - Menos de 95% => UNSTABLE
                                // - Menos de 90% => FAILURE
                                [criticality: 'UNSTABLE', metric: 'LINE', threshold: 95.0],
                                [criticality: 'FAILURE', metric: 'LINE', threshold: 90.0],
                        
                                // Cobertura de ramas:
                                // - Menos de 85% => UNSTABLE
                                // - Menos de 80% => FAILURE
                                [criticality: 'UNSTABLE', metric: 'BRANCH', threshold: 85.0],
                                [criticality: 'FAILURE', metric: 'BRANCH', threshold: 80.0]
                            ]
                        )
                    }
                }

                stage('Rest') {
                    agent { label 'Nodo2' }
                    steps {

                        echo '==- RECUPERANDO ARCHIVOS NODO 2 ==-'
                        unstash 'source-code'
                        catchError(buildResult: 'UNSTABLE', stageResult: 'FAILURE') {
                            bat '''
                                echo ==- CONF. PYTHONPATH NODO 2==-
                                set "PYTHONPATH=%WORKSPACE%"

                                echo ==- DATOS NODO 2 ==-
                                hostname
                                whoami
                                echo WORKSPACE=%WORKSPACE%
                                echo.

                                echo ==- VERSIONES MODULOS NODO 2 ==-
                                java --version
                                python --version
                                python -m pytest --version
                        
                                echo ==- CONF. FLASK NODO 2 ==-
                                set FLASK_APP=app\\api.py
                                set FLASK_ENV=development
                                start "" cmd /c flask run

                                echo ==- CONF. WIREMOCK NODO 2 ==-
                                start "" cmd /c java -jar C:\\Users\\Guarr1or\\Downloads\\Programas\\DevOps\\WireMock\\wiremock-standalone-3.13.2.jar --port 9090 --root-dir test\\wiremock

                                echo ==- ESPERA ARRANQUE SERVICIOS NODO 2 ==-
                                ping 127.0.0.1 -n 11 > nul

                                echo ==- INSTALA MODULOS NODO 2 -==
                                python -m pip install --disable-pip-version-check -U pytest

                                echo ==- EJECUTA PRUEBAS REST NODO 2 -==
                                python -m pytest --junitxml=result-rest.xml test\\rest
                            '''
                        }
                        
                        echo '==- PUBLICA RESULTADOS NODO 2 ==-'
                        junit 'result-rest.xml'
                    }
                }
                                
                stage('StaticsSecur') {
                    agent { label 'Nodo2' }
                    steps {

                        echo '==- RECUPERANDO ARCHIVOS NODO 2 ==-'
                        unstash 'source-code'
                        catchError(buildResult: 'UNSTABLE', stageResult: 'FAILURE') {
                            bat '''
                                echo ==- CONF. PYTHONPATH NODO 2==-
                                set "PYTHONPATH=%WORKSPACE%"

                                echo ==- DATOS NODO 2 ==-
                                hostname
                                whoami
                                echo WORKSPACE=%WORKSPACE%
                                echo.

                                echo ==- VERSIONES MODULOS NODO 2 ==-
                                java --version
                                python --version
                                python -m pytest --version

                                echo ==- INSTALA Y EJECUTA MODULOS NODO 2 -==
                                python -m pip install --disable-pip-version-check -U flake8 bandit
                                flake8 --exit-zero --format=pylint app >result-flake8.out
                                bandit --exit-zero -r . -f custom -o result-bandit.out --msg-template "{abspath}:{line}: [{test_id}] {msg}"
                            '''
                        }

                        echo '==- PUBLICA RESULTADOS NODO 2 ==-'
                        recordIssues(
                            tools: [
                                flake8(
                                    pattern: 'result-flake8.out',
                                    name: 'Flake8'
                                )
                            ],
                            qualityGates: [
                                [threshold: 8, type: 'TOTAL', unstable: true],
                                [threshold: 10, type: 'TOTAL', unstable: false]
                            ]
                        )
                        
                        recordIssues(
                            tools: [
                                pyLint(
                                    pattern: 'result-bandit.out',
                                    name: 'Bandit'
                                )
                            ],
                            qualityGates: [
                                [threshold: 2, type: 'TOTAL', unstable: true],
                                [threshold: 5, type: 'TOTAL', unstable: false]
                            ]
                        )                        

                    }
                }
                
                stage('Performance') {
                    agent { label 'Nodo1' }
                    steps {

                        echo '==- RECUPERANDO ARCHIVOS NODO 1 ==-'
                        unstash 'source-code'
                        catchError(buildResult: 'UNSTABLE', stageResult: 'FAILURE') {
                            bat '''
                                echo ==- DATOS NODO 1 ==-
                                hostname
                                whoami
                                echo WORKSPACE=%WORKSPACE%

                                echo ==- VERSIONES MODULOS NODO 1 ==-
                                java --version

                                echo ==- CONF. FLASK NODO 1 ==-
                                set FLASK_APP=app\\api.py
                                set FLASK_ENV=development
                                start "" cmd /c flask run
                            
                                echo ==- ESPERA ARRANQUE SERVICIOS NODO 1 ==-
                                ping 127.0.0.1 -n 10 > nul
                            
                                echo ==- EJECUTA PRUEBAS PERFORMANCE NODO 1 -==
                                call "C:\\Users\\Guarr1or\\Downloads\\Programas\\DevOps\\JMeter\\apache-jmeter-5.6.3\\bin\\jmeter.bat" -n -t test\\jmeter\\flask.jmx -f -l flask.jtl
                            '''
                        }

                        perfReport( 
                            sourceDataFiles: 'flask.jtl',
                            
                            errorFailedThreshold: 5,
                            errorUnstableThreshold: 1,
                        
                            relativeFailedThresholdPositive: 200,
                            relativeUnstableThresholdPositive: 100                        
                        )

                    }
                }                
            }
        }

    }
}
